   111→    deleteVideo,

   161→      ...videos.map((v) => ({ type: 'video' as const, data: v, createdAt: v.createdAt })),

   192→      reader.onload = (e) => {
   193→        setPreviews((prev) => [...prev, e.target?.result as string])

   237→      description: editAlbumDesc.trim() || undefined,
   238→      category: editAlbumCategory.trim() || undefined,

   399→                </div>

   444→                <div className="flex flex-col items-center justify-center py-20 text-muted-foreground">

   478→                  initial="initial"
   479→                  animate="animate"
   480→                >
   481→                  {mediaItems.map((item, index) => (
   482→                    <motion.div
   483→                      key={item.type === 'photo' ? item.data.id : item.data.id}

   486→                      onClick={() => openLightbox(mediaItems, index)}

   491→                            <img

   493→                              alt={(item.data as Video).title}
   494→                              className="w-full h-full object-cover group-hover:scale-[1.02] transition-transform duration-300"

   497→                            <div className="w-full h-full flex items-center justify-center">

   499→                            </div>
   500→                          )}
   501→                          <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
   502→                            <div className="bg-black/50 rounded-full p-3">
   503→                              <Play className="h-7 w-7 text-white fill-white" />

   505→                          </div>
   506→                          <div className="absolute inset-0 bg-black/0 group-hover:bg-black/20 transition-colors" />

   527→                          <div className="absolute inset-0 bg-black/0 group-hover:bg-black/10 transition-colors" />
   528→                          <div className="absolute top-2 right-2 opacity-0 group-hover:opacity-100 transition-opacity">
   529→                            <Button
   530→                              variant="destructive" size="icon" className="h-8 w-8 rounded-full shadow-lg"
   531→                              onClick={(e) => { e.stopPropagation(); deletePhoto((item.data as Photo).id) }}

   538→                    </motion.div>

   540→                </motion.div>
   541→              ) : (
   542→                /* ============ GRID VIEW ============ */
   543→                <motion.div

   545→                  variants={staggerContainer}
   546→                  initial="initial"
