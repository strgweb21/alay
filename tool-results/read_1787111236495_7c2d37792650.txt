    93→// ==================== MAIN PAGE ====================
    94→
    95→export default function Home() {
    96→  const {
    97→    view,
    98→    albums,
    99→    photos,
   100→    selectedAlbum,
   101→    lightbox,
   102→    loading,
   103→    uploading,
   104→    uploadProgress,
   105→    setView,
   106→    fetchAlbums,
   107→    createAlbum,
   108→    updateAlbum,
   109→    deleteAlbum,
   110→    uploadPhotos,
   111→    deletePhoto,
   112→    openLightbox,
   113→    closeLightbox,
   114→    nextPhoto,
   115→    prevPhoto,
   116→  } = useAlbumStore()
   117→
   118→  // Dialog states
   119→  const [createDialogOpen, setCreateDialogOpen] = useState(false)
   120→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   121→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   122→
   123→  // Form states
   124→  const [newAlbumName, setNewAlbumName] = useState('')
   125→  const [newAlbumDesc, setNewAlbumDesc] = useState('')
   126→  const [editAlbumName, setEditAlbumName] = useState('')
   127→  const [editAlbumDesc, setEditAlbumDesc] = useState('')
   128→  const [editingAlbum, setEditingAlbum] = useState<Album | null>(null)
   129→
   130→  // Upload states
   131→  const [uploadMethod, setUploadMethod] = useState<'local' | 'imgbb'>('local')
   132→  const [selectedFiles, setSelectedFiles] = useState<File[]>([]
   133→  const [previews, setPreviews] = useState<string[]>([])
   134→  const [imgbbApiKey, setImgbbApiKey] = useState('')
   135→  const [showApiKey, setShowApiKey] = useState(false)
   136→  const [isDragOver, setIsDragOver] = useState(false)
   137→  const fileInputRef = useRef<HTMLInputElement>(null)
   138→
   139→  // Load ImgBB API key from localStorage on mount
   140→  useEffect(() => {
   141→    const savedKey = localStorage.getItem('imgbb_api_key')
   142→    if (savedKey) setImgbbApiKey(savedKey)
   143→  }, [])
   144→
   145→  // Fetch albums on mount
   146→  useEffect(() => {
   147→    fetchAlbums()
   148→  }, [fetchAlbums])
   149→
   150→  // Keyboard navigation for lightbox
   151→  useEffect(() => {
   152→    if (!lightbox.open) return
   153→    const handleKeyDown = (e: KeyboardEvent) => {
   154→      if (e.key === 'Escape') closeLightbox()
   155→      if (e.key === 'ArrowRight') nextPhoto()
   156→      if (e.key === 'ArrowLeft') prevPhoto()
   157→    }
   158→    window.addEventListener('keydown', handleKeyDown)
   159→    return () => window.removeEventListener('keydown', handleKeyDown)
   160→  }, [lightbox.open, closeLightbox, nextPhoto, prevPhoto])
   161→
   162→  // Prevent body scroll when lightbox is open
   163→  useEffect(() => {
   164→    if (lightbox.open) {
   165→      document.body.style.overflow = 'hidden'
   166→    } else {
   167→      document.body.style.overflow = ''
   168→    }
   169→    return () => {
   170→      document.body.style.overflow = ''
   171→    }
   172→  }, [lightbox.open])
   173→
   174→  // ---- Handlers ----
   175→
   176→  const handleCreateAlbum = async () => {
   177→    if (!newAlbumName.trim()) return
   178→    const ok = await createAlbum({ name: newAlbumName.trim(), description: newAlbumDesc.trim() || undefined })
   179→    if (ok) {
   180→      setNewAlbumName('')
   181→      setNewAlbumDesc('')
   182→      setCreateDialogOpen(false)
   183→    }
   184→  }
   185→
   186→  const handleOpenEditDialog = (album: Album) => {
   187→    setEditingAlbum(album)
   188→    setEditAlbumName(album.name)
   189→    setEditAlbumDesc(album.description || '')
   190→    setEditDialogOpen(true)
   191→  }
   192→
   193→  const handleEditAlbum = async () => {
   194→    if (!editingAlbum || !editAlbumName.trim()) return
   195→    const ok = await updateAlbum(editingAlbum.id, {
   196→      name: editAlbumName.trim(),
   197→      description: editAlbumDesc.trim() || undefined,
   198→    })
   199→    if (ok) {
   200→      setEditDialogOpen(false)
   201→      setEditingAlbum(null)
   202→    }
   203→  }
   204→
   205→  const handleDeleteAlbum = async (id: string) => {
   206→    await deleteAlbum(id)
   207→  }
   208→
   209→  const handleSelectFiles = (files: FileList | null) => {
   210→    if (!files) return
   211→    const newFiles = Array.from(files).filter((f) => f.type.startsWith('image/'))
   212→    setSelectedFiles((prev) => [...prev, ...newFiles])
   213→
   214→    // Generate previews
   215→    newFiles.forEach((file) => {
   216→      const reader = new FileReader()
   217→      reader.onload = (e) => {
   218→        setPreviews((prev) => [...prev, e.target?.result as string])
   219→      }
   220→      reader.readAsDataURL(file)
   221→    })
   222→  }
   223→
   224→  const handleRemoveFile = (index: number) => {
   225→    setSelectedFiles((prev) => prev.filter((_, i) => i !== index))
   226→    setPreviews((prev) => prev.filter((_, i) => i !== index))
   227→  }
   228→
   229→  const handleUpload = async () => {
   230→    if (!selectedAlbum || selectedFiles.length === 0) return
   231→
   232→    // Save API key to localStorage if using ImgBB
   233→    if (uploadMethod === 'imgbb' && imgbbApiKey.trim()) {
   234→      localStorage.setItem('imgbb_api_key', imgbbApiKey.trim())
   235→    }
   236→
   237→    const ok = await uploadPhotos(
   238→      selectedAlbum.id,
   239→      selectedFiles,
   240→      uploadMethod,
   241→      uploadMethod === 'imgbb' ? imgbbApiKey.trim() : undefined
   242→    )
   243→    if (ok) {
   244→      setSelectedFiles([])
   245→      setPreviews([])
   246→      setUploadDialogOpen(false)
   247→    }
   248→  }
   249→
   250→  const handleDragOver = useCallback((e: React.DragEvent) => {
   251→    e.preventDefault()
   252→    setIsDragOver(true)
   253→  }, [])
   254→
   255→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   256→    e.preventDefault()
   257→    setIsDragOver(false)
   258→  }, [])
   259→
   260→  const handleDrop = useCallback((e: React.DragEvent) => {
   261→    e.preventDefault()
   262→    setIsDragOver(false)
   263→    handleSelectFiles(e.dataTransfer.files)
   264→  }, [])
   265→
   266→  // ---- Render ----
   267→
   268→  return (
   269→    <div className="min-h-screen flex flex-col">
   270→      <main className="flex-1">
   271→        <AnimatePresence mode="wait">
   272→          {view === 'albums' ? (
   273→            <motion.div
   274→              key="albums"
   275→              variants={pageVariants}
   276→              initial="initial"
   277→              animate="animate"
   278→              exit="exit"
   279→              className="w-full max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8"
   280→            >
   281→              <AlbumsView
   282→                albums={albums}
   283→                loading={loading}
   284→                onOpenCreateDialog={() => setCreateDialogOpen(true)}
   285→                onEditAlbum={handleOpenEditDialog}
   286→                onDeleteAlbum={handleDeleteAlbum}
   287→                onSelectAlbum={(album) => setView('photos', album.id)}
   288→              />
   289→            </motion.div>
   290→          ) : (
   291→            <motion.div
   292→              key="photos"
   293→              variants={pageVariants}
   294→              initial="initial"
   295→              animate="animate"
   296→              exit="exit"
   297→              className="w-full max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8"
   298→            >
   299→              <PhotosView
   300→                album={selectedAlbum}
   301→                photos={photos}
   302→                loading={loading}
   303→                onBack={() => setView('albums')}
   304→                onOpenUploadDialog={() => setUploadDialogOpen(true)}
   305→                onDeletePhoto={deletePhoto}
   306→                onOpenLightbox={openLightbox}
   307→              />
   308→            </motion.div>
   309→          )}
   310→        </AnimatePresence>
   311→      </main>
   312→
   313→      {/* Footer */}
   314→      <footer className="mt-auto border-t py-6 text-center text-sm text-muted-foreground">
   315→        <p>&copy; {new Date().getFullYear()} Album Koleksi &mdash; Admin Panel</p>
   316→      </footer>
   317→
   318→      {/* Create Album Dialog */}
   319→      <CreateAlbumDialog
   320→        open={createDialogOpen}
   321→        onOpenChange={setCreateDialogOpen}
   322→        name={newAlbumName}
   323→        description={newAlbumDesc}
   324→        onNameChange={setNewAlbumName}
   325→        onDescriptionChange={setNewAlbumDesc}
   326→        onSubmit={handleCreateAlbum}
   327→      />
   328→
   329→      {/* Edit Album Dialog */}
   330→      <EditAlbumDialog
   331→        open={editDialogOpen}
   332→        onOpenChange={setEditDialogOpen}
   333→        name={editAlbumName}
   334→        description={editAlbumDesc}
   335→        onNameChange={setEditAlbumName}
   336→        onDescriptionChange={setEditAlbumDesc}
   337→        onSubmit={handleEditAlbum}
   338→      />
   339→
   340→      {/* Upload Photos Dialog */}
   341→      <UploadPhotosDialog
   342→        open={uploadDialogOpen}
   343→        onOpenChange={(open) => {
   344→          setUploadDialogOpen(open)
   345→          if (!open) {
   346→            setSelectedFiles([])
   347→            setPreviews([])
   348→          }
   349→        }}
   350→        uploadMethod={uploadMethod}
   351→        onMethodChange={setUploadMethod}
   352→        selectedFiles={selectedFiles}
   353→        previews={previews}
   354→        imgbbApiKey={imgbbApiKey}
   355→        showApiKey={showApiKey}
   356→        isDragOver={isDragOver}
   357→        uploading={uploading}
   358→        uploadProgress={uploadProgress}
   359→        fileInputRef={fileInputRef}
   360→        onApiKeyChange={setImgbbApiKey}
   361→        onToggleShowApiKey={() => setShowApiKey(!showApiKey)}
   362→        onDragOver={handleDragOver}
   363→        onDragLeave={handleDragLeave}
   364→        onDrop={handleDrop}
   365→        onSelectFiles={handleSelectFiles}
   366→        onRemoveFile={handleRemoveFile}
   367→        onUpload={handleUpload}
   368→      />
   369→
   370→      {/* Photo Lightbox */}
   371→      <PhotoLightbox
   372→        lightbox={lightbox}
   373→        onClose={closeLightbox}
   374→        onPrev={prevPhoto}
   375→        onNext={nextPhoto}
   376→      />
   377→    </div>
   378→  )
   379→}

   392→function AlbumsView({
   393→  albums,
   394→  loading,

   397→  onDeleteAlbum,

   432→          </div>
   433→          <h3 className="text-lg font-semibold text-foreground">Belum ada album</h3>

   436→          </p>
   437→          <Button
   438→            onClick={onOpenCreateDialog}
   439→            className="mt-6 bg-emerald-600 hover:bg-emerald-700 text-white gap-2"

   443→          </Button>
   444→        </motion.div>
   445→      )}
   446→
   447→      {/* Albums Grid */}
   448→      {!loading && albums.length > 0 && (
   449→        <motion.div
   450→          variants={staggerContainer}
   451→          initial="initial"
   452→          animate="animate"

   459→                onEdit={onEditAlbum}
   460→                onDelete={onDeleteAlbum}
   461→                onClick={() => onSelectAlbum(album)}
   462→              />
   463→            </motion.div>
   464→          ))}
   465→        </motion.div>
   466→      )}
   467→    </section>
   468→  )
   469→}

   480→function AlbumCard({ album, onEdit, onDelete, onClick }: AlbumCardProps) {

   482→    <motion.div whileHover={{ scale: 1.02 }} transition={{ duration: 0.2 }}>

   509→        <CardContent className="p-4">

   512→            <p className="text-sm text-muted-foreground mt-1 line-clamp-2">{album.description}</p>

   514→          <p className="text-xs text-muted-foreground mt-2">{formatDate(album.createdAt)}</p>

   521→            className="h-8 w-8 p-0 text-muted-foreground hover:text-foreground"

   526→          >
   527→            <Pencil className="h-3.5 w-3.5" />

   529→          <DropdownMenu>

   531→              <Button
   532→                variant="ghost"
   533→                size="sm"
   534→                className="h-8 w-8 p-0 text-muted-foreground hover:text-red-600"
   535→                onClick={(e) => e.stopPropagation()}
   536→              >
   537→                <Trash2 className="h-3.5 w-3.5" />
   538→              </Button>
   539→            </DropdownMenuTrigger>
   540→            <DropdownMenuContent align="end">
   541→              <DropdownMenuItem
   542→                className="text-red-600 focus:text-red-600 focus:bg-red-50"
   543→                onClick={(e) => {
   544→                  e.stopPropagation()
   545→                  onDelete(album.id)
   546→                }}
   547→              >
   548→                <Trash2 className="h-4 w-4 mr-2" />
   549→                Hapus Album
   550→              </DropdownMenuItem>
   551→            </DropdownMenuContent>
   552→          </DropdownMenu>
   553→        </CardFooter>
   554→      </Card>
   555→    </motion.div>
   556→  )
   557→}

   579→}: PhotosViewProps) {
   580→  return (

   618→        <motion.div
