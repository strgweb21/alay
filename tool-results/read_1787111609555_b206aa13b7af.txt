    92→// ==================== MAIN PAGE ====================
    93→
    94→export default function Home() {
    95→  const {
    96→    view,
    97→    albums,
    98→    photos,
    99→    selectedAlbum,
   100→    lightbox,
   101→    loading,
   102→    uploading,
   103→    uploadProgress,
   104→    setView,
   105→    fetchAlbums,
   106→    createAlbum,
   107→    updateAlbum,
   108→    deleteAlbum,
   109→    uploadPhotos,
   110→    deletePhoto,
   111→    openLightbox,
   112→    closeLightbox,
   113→    nextPhoto,
   114→    prevPhoto,
   115→  } = useAlbumStore()
   116→
   117→  // Dialog states
   118→  const [createDialogOpen, setCreateDialogOpen] = useState(false)
   119→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   120→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   121→
   122→  // Form states
   123→  const [newAlbumName, setNewAlbumName] = useState('')
   124→  const [newAlbumDesc, setNewAlbumDesc] = useState('')
   125→  const [editAlbumName, setEditAlbumName] = useState('')
   126→  const [editAlbumDesc, setEditAlbumDesc] = useState('')
   127→  const [editingAlbum, setEditingAlbum] = useState<Album | null>(null)
   128→
   129→  // Upload states
   130→  const [uploadMethod, setUploadMethod] = useState<'local' | 'imgbb'>('local')
   131→  const [selectedFiles, setSelectedFiles] = useState<File[]>([])
   132→  const [previews, setPreviews] = useState<string[]>([])
   133→  const [imgbbApiKey, setImgbbApiKey] = useState('')
   134→  const [showApiKey, setShowApiKey] = useState(false)
   135→  const [isDragOver, setIsDragOver] = useState(false)
   136→  const fileInputRef = useRef<HTMLInputElement>(null)
   137→
   138→  // Load ImgBB API key from localStorage on mount
   139→  useEffect(() => {
   140→    const savedKey = localStorage.getItem('imgbb_api_key')
   141→    if (savedKey) setImgbbApiKey(savedKey)
   142→  }, [])
   143→
   144→  // Fetch albums on mount
   145→  useEffect(() => {
   146→    fetchAlbums()
   147→  }, [fetchAlbums])
   148→
   149→  // Keyboard navigation for lightbox
   150→  useEffect(() => {
   151→    if (!lightbox.open) return
   152→    const handleKeyDown = (e: KeyboardEvent) => {
   153→      if (e.key === 'Escape') closeLightbox()
   154→      if (e.key === 'ArrowRight') nextPhoto()
   155→      if (e.key === 'ArrowLeft') prevPhoto()
   156→    }
   157→    window.addEventListener('keydown', handleKeyDown)
   158→    return () => window.removeEventListener('keydown', handleKeyDown)
   159→  }, [lightbox.open, closeLightbox, nextPhoto, prevPhoto])
   160→
   161→  // Prevent body scroll when lightbox is open
   162→  useEffect(() => {
   163→    if (lightbox.open) {
   164→      document.body.style.overflow = 'hidden'
   165→    } else {
   166→      document.body.style.overflow = ''
   167→    }
   168→    return () => {
   169→      document.body.style.overflow = ''
   170→    }
   171→  }, [lightbox.open])
   172→
   173→  // ---- Handlers ----
   174→
   175→  const handleCreateAlbum = async () => {
   176→    if (!newAlbumName.trim()) return
   177→    const ok = await createAlbum({ name: newAlbumName.trim(), description: newAlbumDesc.trim() || undefined })
   178→    if (ok) {
   179→      setNewAlbumName('')
   180→      setNewAlbumDesc('')
   181→      setCreateDialogOpen(false)
   182→    }
   183→  }
   184→
   185→  const handleOpenEditDialog = (album: Album) => {
   186→    setEditingAlbum(album)
   187→    setEditAlbumName(album.name)
   188→    setEditAlbumDesc(album.description || '')
   189→    setEditDialogOpen(true)
   190→  }
   191→
   192→  const handleEditAlbum = async () => {
   193→    if (!editingAlbum || !editAlbumName.trim()) return
   194→    const ok = await updateAlbum(editingAlbum.id, {
   195→      name: editAlbumName.trim(),
   196→      description: editAlbumDesc.trim() || undefined,
   197→    })
   198→    if (ok) {
   199→      setEditDialogOpen(false)
   200→      setEditingAlbum(null)
   201→    }
   202→  }
   203→
   204→  const handleDeleteAlbum = async (id: string) => {
   205→    await deleteAlbum(id)
   206→  }
   207→
   208→  const handleSelectFiles = (files: FileList | null) => {
   209→    if (!files) return
   210→    const newFiles = Array.from(files).filter((f) => f.type.startsWith('image/'))
   211→    setSelectedFiles((prev) => [...prev, ...newFiles])
   212→
   213→    // Generate previews
   214→    newFiles.forEach((file) => {
   215→      const reader = new FileReader()
   216→      reader.onload = (e) => {
   217→        setPreviews((prev) => [...prev, e.target?.result as string])
   218→      }
   219→      reader.readAsDataURL(file)
   220→    })
   221→  }
   222→
   223→  const handleRemoveFile = (index: number) => {
   224→    setSelectedFiles((prev) => prev.filter((_, i) => i !== index))
   225→    setPreviews((prev) => prev.filter((_, i) => i !== index))
   226→  }
   227→
   228→  const handleClearFiles = () => {
   229→    setSelectedFiles([])
   230→    setPreviews([])
   231→  }
   232→
   233→  const handleClearFiles = () => {
   234→    setSelectedFiles([])
   235→    setPreviews([])
   236→  }
   237→
   238→  const handleUpload = async () => {
   239→    if (!selectedAlbum || selectedFiles.length === 0) return
   240→
   241→    // Save API key to localStorage if using ImgBB
   242→    if (uploadMethod === 'imgbb' && imgbbApiKey.trim()) {
   243→      localStorage.setItem('imgbb_api_key', imgbbApiKey.trim())
   244→    }
   245→
   246→    const ok = await uploadPhotos(
   247→      selectedAlbum.id,
   248→      selectedFiles,
   249→      uploadMethod,
   250→      uploadMethod === 'imgbb' ? imgbbApiKey.trim() : undefined
   251→    )
   252→    if (ok) {
   253→      setSelectedFiles([])
   254→      setPreviews([])
   255→      setUploadDialogOpen(false)
   256→    }
   257→  }
   258→
   259→  const handleDragOver = useCallback((e: React.DragEvent) => {
   260→    e.preventDefault()
   261→    setIsDragOver(true)
   262→  }, [])
   263→
   264→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   265→    e.preventDefault()
   266→    setIsDragOver(false)
   267→  }, [])
   268→
   269→  const handleDrop = useCallback((e: React.DragEvent) => {
   270→    e.preventDefault()
   271→    setIsDragOver(false)
   272→    handleSelectFiles(e.dataTransfer.files)
   273→  }, [])
   274→
   275→  // ---- Render ----
   276→
   277→  return (
   278→    <div className="min-h-screen flex flex-col">
   279→      <main className="flex-1">
   280→        <AnimatePresence mode="wait">
   281→          {view === 'albums' ? (
   282→            <motion.div
   283→              key="albums"
   284→              variants={pageVariants}
   285→              initial="initial"
   286→              animate="animate"
   287→              exit="exit"
   288→              className="w-full max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8"
   289→            >
   290→              <AlbumsView
   291→                albums={albums}
   292→                loading={loading}
   293→                onOpenCreateDialog={() => setCreateDialogOpen(true)}
   294→                onEditAlbum={handleOpenEditDialog}
   295→                onDeleteAlbum={handleDeleteAlbum}
   296→                onSelectAlbum={(album) => setView('photos', album.id)}
   297→              />
   298→            </motion.div>
   299→          ) : (
   300→            <motion.div
   301→              key="photos"
   302→              variants={pageVariants}
   303→              initial="initial"
   304→              animate="animate"
   305→              exit="exit"
   306→              className="w-full max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8"
   307→            >
   308→              <PhotosView
   309→                album={selectedAlbum}
   310→                photos={photos}
   311→                loading={loading}
   312→                onBack={() => setView('albums')}
   313→                onOpenUploadDialog={() => setUploadDialogOpen(true)}
   314→                onDeletePhoto={deletePhoto}
   315→                onOpenLightbox={openLightbox}
   316→              />
   317→            </motion.div>
   318→          )}
   319→        </AnimatePresence>
   320→      </main>
   321→
   322→      {/* Footer */}
   323→      <footer className="mt-auto border-t py-6 text-center text-sm text-muted-foreground">
   324→        <p>&copy; {new Date().getFullYear()} Album Koleksi &mdash; Admin Panel</p>
   325→      </footer>
   326→
   327→      {/* Create Album Dialog */}
   328→      <CreateAlbumDialog
   329→        open={createDialogOpen}
   330→        onOpenChange={setCreateDialogOpen}
   331→        name={newAlbumName}
   332→        description={newAlbumDesc}
   333→        onNameChange={setNewAlbumName}
   334→        onDescriptionChange={setNewAlbumDesc}
   335→        onSubmit={handleCreateAlbum}
   336→      />
   337→
   338→      {/* Edit Album Dialog */}
   339→      <EditAlbumDialog
   340→        open={editDialogOpen}
   341→        onOpenChange={setEditDialogOpen}
   342→        name={editAlbumName}
   343→        description={editAlbumDesc}
   344→        onNameChange={setEditAlbumName}
   345→        onDescriptionChange={setEditAlbumDesc}
   346→        onSubmit={handleEditAlbum}
   347→      />
   348→
   349→      {/* Upload Photos Dialog */}
   350→      <UploadPhotosDialog
   351→        open={uploadDialogOpen}
   352→        onOpenChange={(open) => {
   353→          setUploadDialogOpen(open)
   354→          if (!open) {
   355→            setSelectedFiles([])
   356→            setPreviews([])
   357→          }
   358→        }}
   359→        uploadMethod={uploadMethod}
   360→        onMethodChange={setUploadMethod}
   361→        selectedFiles={selectedFiles}
   362→        previews={previews}
   363→        imgbbApiKey={imgbbApiKey}
   364→        showApiKey={showApiKey}
   365→        isDragOver={isDragOver}
   366→        uploading={uploading}
   367→        uploadProgress={uploadProgress}
   368→        fileInputRef={fileInputRef}
   369→        onApiKeyChange={setImgbbApiKey}
   370→        onToggleShowApiKey={() => setShowApiKey(!showApiKey)}
   371→        onDragOver={handleDragOver}
   372→        onDragLeave={handleDragLeave}
   373→        onDrop={handleDrop}
   374→        onSelectFiles={handleSelectFiles}
   375→        onRemoveFile={handleRemoveFile}
   376→        onClearFiles={handleClearFiles}
   377→        onUpload={handleUpload}
   378→      />
   379→
   380→      {/* Photo Lightbox */}
   381→      <PhotoLightbox
   382→        lightbox={lightbox}
   383→        onClose={closeLightbox}
   384→        onPrev={prevPhoto}
   385→        onNext={nextPhoto}
   386→      />
   387→    </div>
   388→  )
   389→}

   562→          </DropdownMenu>
   563→        </CardFooter>
   564→      </Card>
   565→    </motion.div>
   566→  )
