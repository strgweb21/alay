    92→// ==================== MAIN PAGE ====================
    93→
    94→export default function Home() {
    95→  const {
    96→    view,
    97→    albums,
    98→    photos,
    99→    selectedAlbum,
   100→    lightbox,
   101→    loading,
   102→    uploading,
   103→    uploadProgress,
   104→    setView,
   105→    fetchAlbums,
   106→    createAlbum,
   107→    updateAlbum,
   108→    deleteAlbum,
   109→    uploadPhotos,
   110→    deletePhoto,
   111→    openLightbox,
   112→    closeLightbox,
   113→    nextPhoto,
   114→    prevPhoto,
   115→  } = useAlbumStore()
   116→
   117→  // Dialog states
   118→  const [createDialogOpen, setCreateDialogOpen] = useState(false)
   119→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   120→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   121→
   122→  // Form states
   123→  const [newAlbumName, setNewAlbumName] = useState('')
   124→  const [newAlbumDesc, setNewAlbumDesc] = useState('')
   125→  const [editAlbumName, setEditAlbumName] = useState('')
   126→  const [editAlbumDesc, setEditAlbumDesc] = useState('')
   127→  const [editingAlbum, setEditingAlbum] = useState<Album | null>(null)
   128→
   129→  // Upload states
   130→  const [uploadMethod, setUploadMethod] = useState<'local' | 'imgbb'>('local')
   131→  const [selectedFiles, setSelectedFiles] = useState<File[]>([])
   132→  const [previews, setPreviews] = useState<string[]>([])
   133→  const [imgbbApiKey, setImgbbApiKey] = useState(() => {
   134→    if (typeof window !== 'undefined') {
   135→      return localStorage.getItem('imgbb_api_key') || ''
   136→    }
   137→    return ''
   138→  })
   139→  const [showApiKey, setShowApiKey] = useState(false)
   140→  const [isDragOver, setIsDragOver] = useState(false)
   141→  const fileInputRef = useRef<HTMLInputElement>(null)
   142→
   143→  // Fetch albums on mount
   144→  useEffect(() => {
   145→    fetchAlbums()
   146→  }, [fetchAlbums])
   147→
   148→  // Keyboard navigation for lightbox
   149→  useEffect(() => {
   150→    if (!lightbox.open) return
   151→    const handleKeyDown = (e: KeyboardEvent) => {
   152→      if (e.key === 'Escape') closeLightbox()
   153→      if (e.key === 'ArrowRight') nextPhoto()
   154→      if (e.key === 'ArrowLeft') prevPhoto()
   155→    }
   156→    window.addEventListener('keydown', handleKeyDown)
   157→    return () => window.removeEventListener('keydown', handleKeyDown)
   158→  }, [lightbox.open, closeLightbox, nextPhoto, prevPhoto])
   159→
   160→  // Prevent body scroll when lightbox is open
   161→  useEffect(() => {
   162→    if (lightbox.open) {
   163→      document.body.style.overflow = 'hidden'
   164→    } else {
   165→      document.body.style.overflow = ''
   166→    }
   167→    return () => {
   168→      document.body.style.overflow = ''
   169→    }
   170→  }, [lightbox.open])
   171→
   172→  // ---- Handlers ----
   173→
   174→  const handleCreateAlbum = async () => {
   175→    if (!newAlbumName.trim()) return
   176→    const ok = await createAlbum({ name: newAlbumName.trim(), description: newAlbumDesc.trim() || undefined })
   177→    if (ok) {
   178→      setNewAlbumName('')
   179→      setNewAlbumDesc('')
   180→      setCreateDialogOpen(false)
   181→    }
   182→  }
   183→
   184→  const handleOpenEditDialog = (album: Album) => {
   185→    setEditingAlbum(album)
   186→    setEditAlbumName(album.name)
   187→    setEditAlbumDesc(album.description || '')
   188→    setEditDialogOpen(true)
   189→  }
   190→
   191→  const handleEditAlbum = async () => {
   192→    if (!editingAlbum || !editAlbumName.trim()) return
   193→    const ok = await updateAlbum(editingAlbum.id, {
   194→      name: editAlbumName.trim(),
   195→      description: editAlbumDesc.trim() || undefined,
   196→    })
   197→    if (ok) {
   198→      setEditDialogOpen(false)
   199→      setEditingAlbum(null)
   200→    }
   201→  }
   202→
   203→  const handleDeleteAlbum = async (id: string) => {
   204→    await deleteAlbum(id)
   205→  }
   206→
   207→  const handleSelectFiles = (files: FileList | null) => {
   208→    if (!files) return
   209→    const newFiles = Array.from(files).filter((f) => f.type.startsWith('image/'))
   210→    setSelectedFiles((prev) => [...prev, ...newFiles])
   211→
   212→    // Generate previews
   213→    newFiles.forEach((file) => {
   214→      const reader = new FileReader()
   215→      reader.onload = (e) => {
   216→        setPreviews((prev) => [...prev, e.target?.result as string])
   217→      }
   218→      reader.readAsDataURL(file)
   219→    })
   220→  }
   221→
   222→  const handleRemoveFile = (index: number) => {
   223→    setSelectedFiles((prev) => prev.filter((_, i) => i !== index))
   224→    setPreviews((prev) => prev.filter((_, i) => i !== index))
   225→  }
   226→
   227→  const handleClearFiles = () => {
   228→    setSelectedFiles([])
   229→    setPreviews([])
   230→  }
   231→
   232→  const handleUpload = async () => {
   233→    if (!selectedAlbum || selectedFiles.length === 0) return
   234→
   235→    // Save API key to localStorage if using ImgBB
   236→    if (uploadMethod === 'imgbb' && imgbbApiKey.trim()) {
   237→      localStorage.setItem('imgbb_api_key', imgbbApiKey.trim())
   238→    }
   239→
   240→    const ok = await uploadPhotos(
   241→      selectedAlbum.id,
   242→      selectedFiles,
   243→      uploadMethod,
   244→      uploadMethod === 'imgbb' ? imgbbApiKey.trim() : undefined
   245→    )
   246→    if (ok) {
   247→      setSelectedFiles([])
   248→      setPreviews([])
   249→      setUploadDialogOpen(false)
   250→    }
   251→  }
   252→
   253→  const handleDragOver = useCallback((e: React.DragEvent) => {
   254→    e.preventDefault()
   255→    setIsDragOver(true)
   256→  }, [])
   257→
   258→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   259→    e.preventDefault()
   260→    setIsDragOver(false)
   261→  }, [])
   262→
   263→  const handleDrop = useCallback((e: React.DragEvent) => {
   264→    e.preventDefault()
   265→    setIsDragOver(false)
   266→    handleSelectFiles(e.dataTransfer.files)
   267→  }, [])
   268→
   269→  // ---- Render ----
   270→
   271→  return (
   272→    <div className="min-h-screen flex flex-col">
   273→      <main className="flex-1">
   274→        <AnimatePresence mode="wait">
   275→          {view === 'albums' ? (
   276→            <motion.div
   277→              key="albums"
   278→              variants={pageVariants}
   279→              initial="initial"
   280→              animate="animate"
   281→              exit="exit"
   282→              className="w-full max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8"
   283→            >
   284→              <AlbumsView
   285→                albums={albums}
   286→                loading={loading}
   287→                onOpenCreateDialog={() => setCreateDialogOpen(true)}
   288→                onEditAlbum={handleOpenEditDialog}
   289→                onDeleteAlbum={handleDeleteAlbum}
   290→                onSelectAlbum={(album) => setView('photos', album.id)}
   291→              />
   292→            </motion.div>
   293→          ) : (
   294→            <motion.div
   295→              key="photos"
   296→              variants={pageVariants}
   297→              initial="initial"
   298→              animate="animate"
   299→              exit="exit"
   300→              className="w-full max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8"
   301→            >
   302→              <PhotosView
   303→                album={selectedAlbum}
   304→                photos={photos}
   305→                loading={loading}
   306→                onBack={() => setView('albums')}
   307→                onOpenUploadDialog={() => setUploadDialogOpen(true)}
   308→                onDeletePhoto={deletePhoto}
   309→                onOpenLightbox={openLightbox}
   310→              />
   311→            </motion.div>
   312→          )}
   313→        </AnimatePresence>
   314→      </main>
   315→
   316→      {/* Footer */}
   317→      <footer className="mt-auto border-t py-6 text-center text-sm text-muted-foreground">
   318→        <p>&copy; {new Date().getFullYear()} Album Koleksi &mdash; Admin Panel</p>
   319→      </footer>
   320→
   321→      {/* Create Album Dialog */}
   322→      <CreateAlbumDialog
   323→        open={createDialogOpen}
   324→        onOpenChange={setCreateDialogOpen}
   325→        name={newAlbumName}
   326→        description={newAlbumDesc}
   327→        onNameChange={setNewAlbumName}
   328→        onDescriptionChange={setNewAlbumDesc}
   329→        onSubmit={handleCreateAlbum}
   330→      />
   331→
   332→      {/* Edit Album Dialog */}
   333→      <EditAlbumDialog
   334→        open={editDialogOpen}
   335→        onOpenChange={setEditDialogOpen}
   336→        name={editAlbumName}
   337→        description={editAlbumDesc}
   338→        onNameChange={setEditAlbumName}
   339→        onDescriptionChange={setEditAlbumDesc}
   340→        onSubmit={handleEditAlbum}
   341→      />
   342→
   343→      {/* Upload Photos Dialog */}
   344→      <UploadPhotosDialog
   345→        open={uploadDialogOpen}
   346→        onOpenChange={(open) => {
   347→          setUploadDialogOpen(open)
   348→          if (!open) {
   349→            setSelectedFiles([])
   350→            setPreviews([])
   351→          }
   352→        }}
   353→        uploadMethod={uploadMethod}
   354→        onMethodChange={setUploadMethod}
   355→        selectedFiles={selectedFiles}
   356→        previews={previews}
   357→        imgbbApiKey={imgbbApiKey}
   358→        showApiKey={showApiKey}
   359→        isDragOver={isDragOver}
   360→        uploading={uploading}
   361→        uploadProgress={uploadProgress}
   362→        fileInputRef={fileInputRef}
   363→        onApiKeyChange={setImgbbApiKey}
   364→        onToggleShowApiKey={() => setShowApiKey(!showApiKey)}
   365→        onDragOver={handleDragOver}
   366→        onDragLeave={handleDragLeave}
   367→        onDrop={handleDrop}
   368→        onSelectFiles={handleSelectFiles}
   369→        onRemoveFile={handleRemoveFile}
   370→        onClearFiles={handleClearFiles}
   371→        onUpload={handleUpload}
   372→      />
   373→
   374→      {/* Photo Lightbox */}
   375→      <PhotoLightbox
   376→        lightbox={lightbox}
   377→        onClose={closeLightbox}
   378→        onPrev={prevPhoto}
   379→        onNext={nextPhoto}
   380→      />
   381→    </div>
   382→  )
   383→}

   440→          </p>
   441→          <Button
   442→            onClick={onOpenCreateDialog}

   454→          variants={staggerContainer}
   455→          initial="initial"

   467→            </motion.div>
   468→          ))}
   469→        </motion.div>
   470→      )}
   471→    </section>

   541→                <Trash2 className="h-3.5 w-3.5" />

   545→              <DropdownMenuItem
   546→                className="text-red-600 focus:text-red-600 focus:bg-red-50"

   548→                  e.stopPropagation()

   550→                }}
   551→              >
   552→                <Trash2 className="h-4 w-4 mr-2" />
   553→                Hapus Album
   554→              </DropdownMenuItem>
   555→            </DropdownMenuContent>
   556→          </DropdownMenu>
   557→        </CardFooter>
   558→      </Card>
   559→    </motion.div>
   560→  )
   561→}
